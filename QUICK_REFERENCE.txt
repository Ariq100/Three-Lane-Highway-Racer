// 1. Check if a car can spawn in a lane without overlapping
// Returns: 0 = cannot spawn, 1 = can spawn (empty), 2 = can spawn (follow)
// Usage: Validates lane before spawning new enemy car
int can_spawn_in_lane(GameState& state, Lane lane, double* out_speed);

EXAMPLE:
--------
double follow_speed = 0.0;
int result = can_spawn_in_lane(state, LANE_LEFT, &follow_speed);

if (result == 1) {
    // Empty lane - spawn at normal speed
    new_enemy.speed = 3.5;
} else if (result == 2) {
    // Following a car - use its speed
    new_enemy.speed = follow_speed * 0.95;
} else {
    // Cannot spawn - try another lane
}


// 2. Check if a car can safely change to an adjacent lane
// Returns: true = safe to change, false = would collide
// Usage: Prevents reckless driver and overtaking cars from crashing
bool can_change_lane(GameState& state, double x, double y, 
                     int range_x, int range_y, Lane target_lane);

EXAMPLE:
--------
// Check if reckless car can safely change lanes
if (can_change_lane(state, reckless_car.x_pos, reckless_car.y_pos,
                    reckless_car.get_width(), reckless_car.get_height(),
                    LANE_LEFT)) {
    // Safe - perform lane change
    reckless_car.current_lane = LANE_LEFT;
    reckless_car.target_x = lane_to_x(LANE_LEFT);
} else {
    // Not safe - stay in current lane
}


// 3. Prevent 3-lane blockade - ensures player has escape route
// Returns: true = safe to spawn, false = would create blockade
// Usage: Prevents all three lanes from being blocked simultaneously
bool prevent_3lane_blockade(GameState& state, Lane lane, 
                           double new_car_width, double new_car_height);

EXAMPLE:
--------
if (prevent_3lane_blockade(state, LANE_CENTER, 124, 124)) {
    // Safe to spawn in center lane
    state.active_enemies.add(new_enemy);
} else {
    // Would create blockade - skip this spawn
}

void spawn_enemy(GameState& state) {
    // 1. Generate all car properties first
    int type_roll = random_int(1, 100);
    // ... set type, speed, behavior ...
    
    // 2. Try to find valid lane
    for (int attempts = 0; attempts < 3; attempts++) {
        Lane chosen_lane = static_cast<Lane>(random_int(0, 2));
        
        // 3. Check if we can spawn in this lane
        double followed_speed = 0.0;
        int can_spawn_result = can_spawn_in_lane(state, chosen_lane, 
                                                  &followed_speed);
        
        // 4. Check if it would create blockade
        if (can_spawn_result > 0 && 
            prevent_3lane_blockade(state, chosen_lane, 
                                   new_enemy.get_width(), 
                                   new_enemy.get_height())) {
            // 5. Safe to spawn
            new_enemy.current_lane = chosen_lane;
            
            // 6. Match or go slower than leading car
            if (can_spawn_result == 2) {
                new_enemy.speed = followed_speed * 0.95;
            }
            
            spawned = true;
            break;
        }
    }
    
    state.active_enemies.add(new_enemy);
}

/*
 * ===== NEW DATA STRUCTURE - bounded_array<T, MAX_SIZE> =====
 */

TEMPLATE DEFINITION:
--------------------
template <typename T, int MAX_SIZE>
class bounded_array {
private:
    T data[MAX_SIZE];
    int count;
public:
    void add(const T& value);              // Add element
    T& get(int index);                     // Access element
    void remove_at(int index);             // Remove at index
    int length() const;                    // Current count
    int capacity() const;                  // Max capacity
    void clear();                          // Clear all
    T& operator[](int index);              // Bracket operator
};

USAGE:
------
bounded_array<EnemyCar, 30> enemies;  // Max 30 enemies

enemies.add(new_car);                 // Add car
EnemyCar& car = enemies.get(0);       // Get by index
enemies.remove_at(0);                 // Remove by index
int count = enemies.length();         // How many cars
int max = enemies.capacity();         // Max capacity (30)
enemies.clear();                      // Remove all

/*
 * ===== LANE CHANGE VALIDATION RULES =====
 */

For RECKLESS and OVERTAKING cars:

1. Spatial Check: No AABB overlap with target lane car
2. Speed Check:
   - If car ahead (above): Must be faster or equal speed
   - If car behind (below): Must be slower or equal speed
3. Distance Check: Car must be within 200 pixels vertically
4. Call: can_change_lane() validates all rules automatically


/*
 * ===== 3-LANE BLOCKADE PREVENTION =====
 */

The system monitors traffic to prevent impossible situations:

Blockade Scenario:
  Lane 1: [Car A] ---- (gap)
  Lane 2: [Car B] ---- (gap)
  Lane 3:    (empty)

Problem: Player cannot overtake - all lanes are blocked

Solution:
  - prevent_3lane_blockade() detects this
  - Prevents spawn in Lane 3 when A and B are within 150 pixels
  - Waits for Car A or B to pass further down the road
  - Creates guaranteed escape route for player
